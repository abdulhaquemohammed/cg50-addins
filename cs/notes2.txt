=== Searching code
# Linear
- def linear(a, t):
-     for i in range(len(a)):
-         if a[i] == t: return i
-     return -1
# Binary
- def binary(a, t):
-     lo, hi = 0, len(a) - 1
-     while lo <= hi:
-         mid = (lo + hi) // 2
-         if a[mid] == t: return mid
-         elif a[mid] < t: lo = mid + 1
-         else: hi = mid - 1
-     return -1
=== Sorting code
# Bubble
- for p in range(len(a) - 1):
-     swapped = False
-     for i in range(len(a) - 1 - p):
-         if a[i] > a[i+1]:
-             a[i], a[i+1] = a[i+1], a[i]
-             swapped = True
-     if not swapped: break
# Insertion
- for i in range(1, len(a)):
-     item = a[i]; j = i - 1
-     while j >= 0 and a[j] > item:
-         a[j+1] = a[j]; j -= 1
-     a[j+1] = item
