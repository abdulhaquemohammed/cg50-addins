cmake_minimum_required(VERSION 3.15)
project(CSNotes)
include(GenerateG3A)
include(Fxconv)
find_package(Gint 2.9 REQUIRED)
set(C ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
set(SOURCES src/main.c ${C}/num.c ${C}/expr.c ${C}/la.c ${C}/ui.c ${C}/util.c ${C}/stats.c ${C}/plat_gint.c ${CMAKE_CURRENT_SOURCE_DIR}/../../cs/notes.c ${CMAKE_CURRENT_SOURCE_DIR}/../../cs/boolean.c)
set(ASSETS_cg assets-cg/calcfont.png)
fxconv_declare_assets(${ASSETS_cg} WITH_METADATA)
add_executable(cs ${SOURCES} ${ASSETS_cg})
target_compile_options(cs PRIVATE -Wall -O2 -Wno-unused-function)
target_link_libraries(cs Gint::Gint -lm)
generate_g3a(TARGET cs OUTPUT "CSNotes.g3a" NAME "CS Notes" ICONS assets-cg/icon-uns.png assets-cg/icon-sel.png)
